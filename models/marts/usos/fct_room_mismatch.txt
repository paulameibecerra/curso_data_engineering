WITH base AS (
    SELECT
        booking_id,
        dr.room_id,
        hotel_id,
        reserved_room_type,
        assigned_room_type,
        CASE WHEN reserved_room_type != assigned_room_type THEN 1 ELSE 0 END AS is_mismatch
    FROM {{ ref('fact_booking') }} fb
    LEFT JOIN {{ ref('dim_room') }} dr ON fb.room_id = dr.room_id
),

hotel_agg AS (
    SELECT
        hotel_id,
        COUNT(*) AS total_bookings,
        SUM(is_mismatch) AS mismatches,
        ROUND(SUM(is_mismatch) / COUNT(*), 4) AS mismatch_rate
    FROM base
    GROUP BY hotel_id
),

transition_matrix AS (
    SELECT
        reserved_room_type,
        assigned_room_type,
        COUNT(*) AS transitions
    FROM base
    GROUP BY 1,2
)

SELECT
    tm.*,
    ha.mismatch_rate
FROM transition_matrix tm
LEFT JOIN hotel_agg ha
    ON 1=1;
