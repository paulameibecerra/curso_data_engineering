WITH base AS (
    SELECT
        fb.client_id,
        fb.booking_id,
        fb.total_of_special_requests,
        fb.adr,
        fb.days_in_waiting_list,
        pc.adults,
        pc.children,
        pc.babies,
        st.total_stays_nights,
        ar.arrival_date,
        cl.email,
        cl.name,
        cr.country
    FROM {{ ref('fact_booking') }} fb
    LEFT JOIN {{ ref('dim_persons_count') }} pc ON fb.person_count_id = pc.person_count_id
    LEFT JOIN {{ ref('dim_stays') }} st ON fb.stays_id = st.stays_id
    LEFT JOIN {{ ref('dim_arrival_date') }} ar ON fb.arrival_id = ar.arrival_id
    LEFT JOIN {{ ref('dim_client') }} cl ON fb.client_id = cl.client_id
    LEFT JOIN {{ref('dim_country')}} cr ON fb.country_id = cr.country_id
),

agg AS (
    SELECT
        client_id,
        MAX(name) AS name,
        MAX(email) AS email,
        MAX(country) AS country,
        SUM(total_of_special_requests) AS total_special_requests,
        AVG(adr) AS avg_adr,
        MAX(days_in_waiting_list) AS max_waiting_days,
        SUM(children + babies) AS total_minors,
        SUM(adults) AS total_adults
    FROM base
    GROUP BY client_id
),

scored AS (
    SELECT
        *,
        CASE 
            WHEN total_minors >= 1 AND total_special_requests >= 1 THEN 'HIGH'
            WHEN total_special_requests >= 2 THEN 'MEDIUM'
            ELSE 'LOW'
        END AS support_priority
    FROM agg
)

SELECT * FROM scored;
